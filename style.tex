  {\large\raggedright}

  {}{0em}
  {}

% =========================
% Billryan 自定义命令 (resume.cls 第64-122行)
% =========================

% 日期章节 (左标题，右日期)
\newcommand{\datedsection}[2]{%
  \section[#1]{#1 \hfill #2}%
}

% 日期子章节 (公司/学校 + 日期)
\newcommand{\datedsubsection}[2]{%
  \subsection[#1]{#1 \hfill #2}%
}

% 日期行 (单行，两端对齐)
\newcommand{\datedline}[2]{%
  {\par #1 \hfill #2 \par}%
}

% 姓名居中，小大写
\newcommand{\name}[1]{%
  \centerline{\Huge\scshape{#1}}%
  \vspace{1.25ex}%
}

% 联系信息居中
\newcommand{\basicInfo}[1]{%
  \centerline{\sffamily\large{#1}}%
  \vspace{1.5ex}%
}

% 职位/角色说明
\newcommand{\role}[2]{%
  {\par \textit{#1} ~ #2 \par}%
  \vspace{0.85ex}%
}

% 联系方式图标命令 (简化版，不使用 FontAwesome)
\newcommand{\email}[1]{\href{mailto:#1}{#1}}
\newcommand{\phone}[1]{#1}
\newcommand{\linkedin}[2][]{%
  \ifthenelse{\isempty{#1}}%
    {\href{#2}{#2}}
    {\href{#2}{#1}}}
\newcommand{\github}[2][]{%
  \ifthenelse{\isempty{#1}}%
    {\href{#2}{#2}}
    {\href{#2}{#1}}}
\newcommand{\homepage}[2][]{%
  \ifthenelse{\isempty{#1}}%
    {\href{#2}{#2}}
    {\href{#2}{#1}}}
