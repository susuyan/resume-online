\usepackage{xcolor}
% 现代化配色方案 - 深靛蓝主色 + 暖金色强调
\definecolor{PrimaryColor}{HTML}{1A365D}
\definecolor{SecondaryColor}{HTML}{2C5282}
\definecolor{AccentColor}{HTML}{C05621}
\definecolor{TextColor}{HTML}{2D3748}
\definecolor{LightGray}{HTML}{718096}
\definecolor{UltraLight}{HTML}{EDF2F7}
\definecolor{BorderColor}{HTML}{E2E8F0}

\usepackage[
  top=0.7cm,
  bottom=0.7cm,
  left=1.0cm,
  right=1.0cm
]{geometry}

\usepackage{setspace}
\setstretch{1.1}

\usepackage{titlesec}
\setcounter{secnumdepth}{0}

% H1: 姓名 - 更精致的设计
\usepackage{sectsty}
\sectionfont{\centering\Huge\bfseries\color{PrimaryColor}}

\titlespacing{\section}{0pt}{0pt}{6pt}

% H2: 章节标题 - 左侧色块+文字设计
\titleformat{\subsection}
  {\Large\bfseries\color{PrimaryColor}}
  {}
  {0em}
  {}
  [\vspace{-2pt}\textcolor{AccentColor}{\rule{0.6\linewidth}{1.2pt}}]

\titlespacing{\subsection}{0pt}{12pt}{6pt}

% H3: 子标题 - 公司/职位名
\titleformat{\subsubsection}
  {\normalsize\bfseries\color{SecondaryColor}}
  {}
  {0em}
  {}

\titlespacing{\subsubsection}{0pt}{5pt}{2pt}

% 项目符号列表 - 使用实心圆点
\usepackage{enumitem}
\setlist[itemize]{
  leftmargin=1.3em,
  itemsep=1pt,
  topsep=3pt,
  parsep=0pt,
  label={\textcolor{AccentColor}{\textbullet}}
}

% 嵌套列表也使用橙色圆点
\setlist[itemize,2]{
  leftmargin=1.2em,
  itemsep=1pt,
  topsep=2pt,
  parsep=0pt,
  label={\textcolor{AccentColor}{\textbullet}}
}
