% =========================
% Resume Style Definition
% =========================

\usepackage{xcolor}
\definecolor{PrimaryColor}{HTML}{000000} % 恢复黑色或深色

\usepackage[
  top=1.5cm,
  bottom=1.5cm,
  left=2cm,
  right=2cm
]{geometry}

\usepackage{setspace}
\setstretch{1.1}

\usepackage{titlesec}
\setcounter{secnumdepth}{0}

% H1: 姓名
\titleformat{\section}
  {\centering\Huge\bfseries}
  {}
  {0em}
  {}

\titlespacing{\section}{0pt}{0pt}{10pt}

% H2: 章节标题 (下划线)
\titleformat{\subsection}
  {\Large\bfseries}
  {}
  {0em}
  {}
  [\titlerule]

\titlespacing{\subsection}{0pt}{10pt}{5pt}

% H3: 子标题 (加粗，主题色 - 这里暂时用黑色)
\titleformat{\subsubsection}
  {\large\bfseries}
  {}
  {0em}
  {}

\titlespacing{\subsubsection}{0pt}{5pt}{2pt}

\usepackage{enumitem}
\setlist[itemize]{
  leftmargin=1.5em,
  itemsep=2pt,
  topsep=2pt
}

\usepackage[hidelinks]{hyperref}
\setlength{\parindent}{0pt}
