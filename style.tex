% 页面设置
\usepackage[
  a4paper,
  top=1cm,
  bottom=1cm,
  left=1.2cm,
  right=1.2cm
]{geometry}

% 基础包
\usepackage{xcolor}
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage{hyperref}
\usepackage{fontawesome5}
\usepackage{tabularx}
\usepackage{array}

% 颜色定义
\definecolor{primary}{HTML}{2C3E50}
\definecolor{secondary}{HTML}{34495E}
\definecolor{accent}{HTML}{3498DB}
\definecolor{text}{HTML}{2C3E50}
\definecolor{lightgray}{HTML}{95A5A6}
\definecolor{headerblue}{HTML}{1E3A5F}

% 字体设置
\usepackage{fontspec}
\setmainfont{Noto Sans CJK SC}
\setsansfont{Noto Sans CJK SC}

% 取消章节编号
\setcounter{secnumdepth}{0}

% 标题样式
\titleformat{\section}
  {\Large\bfseries\color{headerblue}}
  {}
  {0em}
  {}
  [\vspace{-0.5em}\textcolor{accent}{\rule{\linewidth}{1.5pt}}]

\titlespacing*{\section}{0pt}{12pt}{8pt}

\titleformat{\subsection}
  {\large\bfseries\color{primary}}
  {}
  {0em}
  {}

\titlespacing*{\subsection}{0pt}{10pt}{4pt}

\titleformat{\subsubsection}
  {\normalsize\bfseries\color{secondary}}
  {}
  {0em}
  {}

\titlespacing*{\subsubsection}{0pt}{6pt}{2pt}

% 列表样式
\setlist[itemize]{
  leftmargin=1.2em,
  itemsep=2pt,
  topsep=4pt,
  parsep=0pt,
  label={\textcolor{accent}{\textbullet}}
}

% 段落设置
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% 超链接
\hypersetup{
  colorlinks=true,
  linkcolor=accent,
  urlcolor=accent
}

% 防止溢出
\sloppy
\tolerance=1000
\emergencystretch=3em

% 自定义命令
\newcommand{\resumeheader}[4]{%
  \begin{center}
    {\Huge\bfseries\color{headerblue} #1}\\[6pt]
    {\large\color{secondary} #2}\\[4pt]
    {\color{lightgray} #3}
  \end{center}
  \vspace{8pt}
}

\newcommand{\jobtitle}[3]{%
  \textbf{\color{primary}#1} \hfill {\color{lightgray}#2}\\
  {\color{secondary}#3}
  \vspace{2pt}
}
