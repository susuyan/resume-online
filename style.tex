% =========================
% Resume Style - Billryan Inspired
% =========================

\usepackage{xcolor}
\usepackage{microtype}

% 专业配色方案
\definecolor{PrimaryColor}{HTML}{2C3E50}
\definecolor{AccentColor}{HTML}{3498DB}
\definecolor{TextColor}{HTML}{333333}
\definecolor{LightGray}{HTML}{7F8C8D}
\definecolor{LineColor}{HTML}{BDC3C7}

% 页面边距 - 紧凑单页
\usepackage[
  a4paper,
  left=0.75in,
  right=0.75in,
  top=0.6in,
  bottom=0.5in,
  nohead
]{geometry}

% 行高
\usepackage{setspace}
\setstretch{1.05}

\usepackage{titlesec}
\usepackage{enumitem}
\setcounter{secnumdepth}{0}

% =========================
% 章节标题样式 - 小大写+横线
% =========================
\titleformat{\section}
  {\Large\scshape\raggedright\color{PrimaryColor}}
  {}
  {0em}
  {}
  [\vspace{-0.5em}\textcolor{LineColor}{\rule{\linewidth}{0.5pt}}]

\titlespacing*{\section}{0pt}{1.2em}{0.8em}

% =========================
% 子章节（公司/项目）样式
% =========================
\titleformat{\subsection}
  {\large\bfseries\color{PrimaryColor}}
  {}
  {0em}
  {}

\titlespacing*{\subsection}{0pt}{0.8em}{0.3em}

% =========================
% 小标题（职位/日期）样式
% =========================
\titleformat{\subsubsection}
  {\normalsize\color{TextColor}}
  {}
  {0em}
  {}

\titlespacing*{\subsubsection}{0pt}{0.3em}{0.2em}

% =========================
% 列表样式 - 超紧凑
% =========================
\setlist[itemize]{
  noitemsep,
  topsep=0.2em,
  leftmargin=1.2em,
  labelsep=0.5em,
  label={\textcolor{AccentColor}{\textbullet}}
}

% =========================
% 超链接
% =========================
\usepackage[hidelinks]{hyperref}
\urlstyle{tt}

% =========================
% 排版优化
% =========================
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\sloppy
\tolerance=3000
\emergencystretch=3em
\hbadness=10000

% =========================
% 自定义命令
% =========================

% 姓名居中
\newcommand{\centeredname}[1]{%
  \begin{center}
    {\Huge\scshape\color{PrimaryColor}#1}
  \end{center}
  \vspace{0.5em}
}

% 联系信息居中
\newcommand{\contactinfo}[1]{%
  \begin{center}
    {\normalsize\color{LightGray}#1}
  \end{center}
  \vspace{0.8em}
}

% 日期行（左标题，右日期）
\newcommand{\datedline}[2]{%
  \noindent\textbf{#1} \hfill \textcolor{LightGray}{#2}\\
}

% =========================
% 表格工具
% =========================
\usepackage{array}
\usepackage{tabularx}
