\usepackage{xcolor}
\definecolor{PrimaryColor}{HTML}{2C3E50}
\definecolor{SecondaryColor}{HTML}{34495E}
\definecolor{AccentColor}{HTML}{3498DB}
\definecolor{TextColor}{HTML}{333333}
\definecolor{LightGray}{HTML}{7F8C8D}

\usepackage[
  top=1.2cm,
  bottom=1.2cm,
  left=1.5cm,
  right=1.5cm
]{geometry}

\usepackage{setspace}
\setstretch{1.15}

\usepackage{titlesec}
\setcounter{secnumdepth}{0}

% H1: 姓名
\titleformat{\section}
  {\centering\Huge\bfseries\color{PrimaryColor}}
  {}
  {0em}
  {}

\titlespacing{\section}{0pt}{0pt}{10pt}

% H2: 章节标题 (下划线)
\titleformat{\subsection}
  {\Large\bfseries\color{PrimaryColor}}
  {}
  {0em}
  {}
  [\titlerule]

\titlespacing{\subsection}{0pt}{10pt}{5pt}

% H3: 子标题
\titleformat{\subsubsection}
  {\large\bfseries\color{SecondaryColor}}
  {}
  {0em}
  {}

\titlespacing{\subsubsection}{0pt}{5pt}{2pt}

\usepackage{enumitem}
\setlist[itemize]{
  leftmargin=1.5em,
  itemsep=2pt,
  topsep=2pt
}

\usepackage[hidelinks]{hyperref}
\setlength{\parindent}{0pt}

% 表格样式
\usepackage{booktabs}
\usepackage{array}
\renewcommand{\arraystretch}{1.3}

% 防止文字溢出
\sloppy
\tolerance=1000
\emergencystretch=3em
