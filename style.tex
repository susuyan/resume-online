% Resume Style Definition - Professional CV Style
% =========================

\usepackage{xcolor}
\definecolor{PrimaryColor}{HTML}{1a1a1a}
\definecolor{SecondaryColor}{HTML}{4a4a4a}
\definecolor{AccentColor}{HTML}{2c5282}
\definecolor{LightGray}{HTML}{718096}
\definecolor{BorderColor}{HTML}{e2e8f0}

\usepackage[
  a4paper,
  top=1.2cm,
  bottom=1.2cm,
  left=1.5cm,
  right=1.5cm
]{geometry}

\usepackage{setspace}
\setstretch{1.1}

\usepackage{titlesec}
\setcounter{secnumdepth}{0}

% H1: 姓名 - 居中大字
\titleformat{\section}
  {\centering\LARGE\bfseries\color{PrimaryColor}}
  {}
  {0em}
  {}

\titlespacing{\section}{0pt}{0pt}{8pt}

% H2: 章节标题 - 带下划线
\titleformat{\subsection}
  {\large\bfseries\color{PrimaryColor}}
  {}
  {0em}
  {}
  [\vspace{2pt}{\color{BorderColor}\titlerule[0.8pt]}]

\titlespacing{\subsection}{0pt}{12pt}{8pt}

% H3: 公司/项目名称 - 使用subsubsection
\titleformat{\subsubsection}
  {\normalsize\bfseries\color{PrimaryColor}}
  {}
  {0em}
  {}

\titlespacing{\subsubsection}{0pt}{10pt}{4pt}

% 段落样式 - 紧凑
\usepackage{enumitem}
\setlist[itemize]{
  leftmargin=1.2em,
  itemsep=2pt,
  topsep=3pt,
  parsep=0pt,
  label={\color{AccentColor}\textbullet}
}

% 超链接
\usepackage[hidelinks]{hyperref}

% 段落设置
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% 字体 - 使用系统自带中文字体
\usepackage{fontspec}
\setmainfont{Noto Sans CJK SC}[
  BoldFont=Noto Sans CJK SC Bold,
  UprightFont=Noto Sans CJK SC Regular
]
\setsansfont{Noto Sans CJK SC}[
  BoldFont=Noto Sans CJK SC Bold,
  UprightFont=Noto Sans CJK SC Regular
]

% 确保中文字体正确显示
\usepackage{xeCJK}
\xeCJKsetup{xeCJKactive=false}

% 页脚
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[C]{\color{LightGray}\small 余胜民 · 全栈软件服务交付工程师}

% 防止文字溢出
\sloppy
\tolerance=1000
\emergencystretch=3em

% 表格样式
\usepackage{array}
\usepackage{booktabs}
\setlength{\tabcolsep}{4pt}
\renewcommand{\arraystretch}{1.1}
