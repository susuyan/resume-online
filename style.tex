\usepackage{xcolor}
\definecolor{PrimaryColor}{HTML}{1a1a1a}
\definecolor{SecondaryColor}{HTML}{4a4a4a}
\definecolor{AccentColor}{HTML}{2c5282}
\definecolor{LightGray}{HTML}{718096}
\definecolor{BorderColor}{HTML}{e2e8f0}

\usepackage[
  a4paper,
  top=1.5cm,
  bottom=1.5cm,
  left=1.5cm,
  right=1.5cm
]{geometry}

\usepackage{setspace}
\setstretch{1.15}

\usepackage{titlesec}
\setcounter{secnumdepth}{0}

% H1: 姓名 - 居中大字
\titleformat{\section}
  {\centering\LARGE\bfseries\color{PrimaryColor}}
  {}
  {0em}
  {}

\titlespacing{\section}{0pt}{0pt}{6pt}

% H2: 章节标题 - 带下划线
\titleformat{\subsection}
  {\large\bfseries\color{PrimaryColor}}
  {}
  {0em}
  {}
  [\vspace{1pt}{\color{BorderColor}\titlerule[0.5pt]}]

\titlespacing{\subsection}{0pt}{10pt}{6pt}

% H3: 公司/项目名称
\titleformat{\subsubsection}
  {\normalsize\bfseries\color{PrimaryColor}}
  {}
  {0em}
  {}

\titlespacing{\subsubsection}{0pt}{8pt}{2pt}

% 段落样式
\usepackage{enumitem}
\setlist[itemize]{
  leftmargin=1em,
  itemsep=1pt,
  topsep=2pt,
  parsep=0pt,
  label={\color{AccentColor}\textbullet}
}

% 超链接
\usepackage[hidelinks]{hyperref}

% 段落设置
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% 表格样式
\usepackage{array}
\usepackage{booktabs}
\setlength{\tabcolsep}{6pt}
\renewcommand{\arraystretch}{1.2}

% 字体
\usepackage{fontspec}
\setmainfont{Noto Sans CJK SC}
\setsansfont{Noto Sans CJK SC}

% 页脚
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
