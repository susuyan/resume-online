% =========================
% Resume Style Definition
% =========================

% ---------- Colors ----------
\usepackage{xcolor}
\definecolor{PrimaryColor}{HTML}{2E3A59} % 深蓝灰

% ---------- Page Layout ----------
\usepackage[
  top=1.5cm,
  bottom=1.5cm,
  left=2cm,
  right=2cm
]{geometry}

% ---------- Line Spacing ----------
\usepackage{setspace}
\setstretch{1.05}

% ---------- Section Style ----------
\usepackage{titlesec}

\titleformat{\section}
  {\color{PrimaryColor}\scshape\large\bfseries}
  {}
  {0em}
  {}
  [\color{PrimaryColor}\titlerule]

\titlespacing{\section}{0pt}{10pt}{6pt}

% ---------- Name Title ----------
\newcommand{\resumetitle}[1]{
  \begin{center}
    {\Huge\bfseries\color{PrimaryColor} #1}
  \end{center}
  \vspace{6pt}
}

% ---------- Sub Entry ----------
\newcommand{\resumesubentry}[2]{
  \textbf{#1} \hfill \textit{#2}\\
}

% ---------- Item List ----------
\usepackage{enumitem}
\setlist[itemize]{
  leftmargin=1.2em,
  itemsep=2pt,
  topsep=2pt
}

% ---------- Links ----------
\usepackage[hidelinks]{hyperref}

% ---------- Paragraph ----------
\setlength{\parindent}{0pt}
