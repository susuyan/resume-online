% =========================
% Resume Style - Final Optimized for One Page
% Full-Stack Software Engineer Resume
% =========================

% =========================
% 基础包
% =========================
\usepackage{xcolor}
\usepackage[top=0.6cm,bottom=0.6cm,left=0.8cm,right=0.8cm]{geometry}
\usepackage{setspace}
\usepackage{array}
\usepackage{tabularx}
\usepackage{booktabs}

% =========================
% 专业配色方案
% =========================
\definecolor{PrimaryColor}{HTML}{1A365D}
\definecolor{SecondaryColor}{HTML}{2C5282}
\definecolor{AccentColor}{HTML}{C05621}
\definecolor{LightGray}{HTML}{718096}
\definecolor{TextColor}{HTML}{2D3748}
\definecolor{BorderColor}{HTML}{E2E8F0}
\definecolor{HeaderBg}{HTML}{EDF2F7}

% =========================
% 页面布局 - 单页紧凑优化
% =========================
\setstretch{1.05}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\pagestyle{empty}

% =========================
% 字体配置 - XeLaTeX优化
% =========================
\usepackage{fontspec}
\usepackage{xeCJK}

% 英文字体 - 使用 Latin Modern Sans（texlive 标准字体）
\setsansfont{Latin Modern Sans}
\renewcommand{\familydefault}{\sfdefault}

% 中文字体
\setCJKmainfont{Noto Sans CJK SC}[
  BoldFont=Noto Sans CJK SC Bold,
  ItalicFont=Noto Sans CJK SC Light
]

% 中文换行优化
\XeTeXlinebreaklocale "zh"
\XeTeXlinebreakskip = 0pt plus 0.1pt

% =========================
% 页面优化
% =========================
\sloppy
\tolerance=1000
\emergencystretch=1em
\hyphenpenalty=10000
\exhyphenpenalty=10000
\clubpenalty=10000
\widowpenalty=10000

% =========================
% 兼容性保障（Pandoc生成所需）
% =========================
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}%
}

\providecommand{\subtitle}[1]{}

% =========================
% 章节标题样式
% =========================
\let\oldsection\section
\renewcommand{\section}[1]{%
  \vspace{6pt}%
  {\color{PrimaryColor}\bfseries\normalsize #1}%
  \vspace{2pt}%
  {\color{BorderColor}\hrule height 0.8pt}%
  \vspace{3pt}%
}

\let\oldsubsection\subsection
\renewcommand{\subsection}[1]{%
  \vspace{4pt}%
  {\bfseries\color{SecondaryColor}#1}%
  \vspace{2pt}%
}

% =========================
% 表格样式 - 工作经历时间线
% =========================
\renewcommand{\arraystretch}{1.1}
\newcolumntype{L}[1]{>{\raggedright\arraybackslash}p{#1}}
\newcolumntype{C}[1]{>{\centering\arraybackslash}p{#1}}
\newcolumntype{R}[1]{>{\raggedleft\arraybackslash}p{#1}}
\arrayrulecolor{BorderColor}

% =========================
% 列表样式 - 紧凑
% =========================
\renewcommand{\labelitemi}{{\color{AccentColor}\textbullet}}
\setlength{\leftmargini}{1.2em}
\setlength{\itemsep}{1pt}
\setlength{\parsep}{0pt}
\setlength{\topsep}{2pt}

% =========================
% 超链接样式
% =========================
\usepackage[hidelinks]{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=SecondaryColor,
  urlcolor=AccentColor
}
