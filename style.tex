\usepackage{titlesec}
\usepackage{xcolor}
\usepackage{enumitem}
\usepackage{geometry}

% Define Colors
\definecolor{headerblue}{HTML}{2E5DAC}
\definecolor{namegray}{HTML}{466AA0}

% Section Formatting (H2)
% Blue, Bold, Uppercase, with a horizontal rule below
\titleformat{\section}
  {\color{headerblue}\Large\bfseries\uppercase} % Format
  {} % Label
  {0em} % Sep
  {} % Before-code
  [\titlerule] % After-code (horizontal rule)

% Subsection Formatting (H3) - Used for Job Titles
% We want the date to be right-aligned. 
% In Markdown headers, we can't easily put \hfill. 
% Strategy: We will assume H3 is just the Company Name, and we put the date in the next line or handle it manually in MD.
% OR, we keep H3 simple and use bold text in MD.
% Let's stick to standard formatting for H3 but ensure it's compact.
\titleformat{\subsection}
  {\large\bfseries}
  {}
  {0em}
  {}

% Spacing
\titlespacing*{\section}{0pt}{1.5ex plus 1ex minus .2ex}{1ex plus .2ex}
\titlespacing*{\subsection}{0pt}{1ex plus .5ex minus .2ex}{0.5ex plus .2ex}

% List Formatting
\setlist[itemize]{leftmargin=1.5em, itemsep=2pt, topsep=2pt}

% Hyperlink colors (already set in pandoc args, but ensuring here)
\hypersetup{
    colorlinks=true,
    linkcolor=headerblue,
    urlcolor=headerblue
}
