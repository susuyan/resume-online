% =========================
% Resume Style - One Page Professional

\usepackage{xcolor}
% 专业配色方案
\definecolor{PrimaryColor}{HTML}{1A365D}
\definecolor{SecondaryColor}{HTML}{2C5282}
\definecolor{AccentColor}{HTML}{C05621}
\definecolor{TextColor}{HTML}{2D3748}
\definecolor{LightGray}{HTML}{718096}
\definecolor{BorderColor}{HTML}{E2E8F0}

% 紧凑边距 - 为单页优化
\usepackage[
  top=0.5cm,
  bottom=0.5cm,
  left=0.6cm,
  right=0.6cm
]{geometry}

% 行高 - 紧凑但可读
\usepackage{setspace}
\setstretch{1.0}

\usepackage{titlesec}
\setcounter{secnumdepth}{0}

% =========================
% 头部样式 - 姓名和职位
% =========================
\usepackage{sectsty}
\sectionfont{\centering\LARGE\bfseries\color{PrimaryColor}}

\titlespacing{\section}{0pt}{0pt}{4pt}

% =========================
% 章节标题样式
% =========================
\titleformat{\subsection}
  {\normalsize\bfseries\color{PrimaryColor}\uppercase}
  {}
  {0em}
  {}
  [\vspace{-4pt}\textcolor{AccentColor}{\rule{\linewidth}{1pt}}]

\titlespacing{\subsection}{0pt}{6pt}{4pt}

% =========================
% 公司/职位标题
% =========================
\titleformat{\subsubsection}
  {\small\bfseries\color{SecondaryColor}}
  {}
  {0em}
  {}

\titlespacing{\subsubsection}{0pt}{3pt}{0pt}

% =========================
% 列表样式 - 超紧凑
% =========================
\usepackage{enumitem}
\setlist[itemize]{
  leftmargin=1.2em,
  itemsep=0pt,
  topsep=2pt,
  parsep=0pt,
  labelsep=0.3em,
  label={\textcolor{AccentColor}{\textbullet}}
}

% 嵌套列表
\setlist[itemize,2]{
  leftmargin=1em,
  itemsep=0pt,
  topsep=1pt,
  parsep=0pt,
  label={\textcolor{AccentColor}{\scriptsize$\circ$}}
}

% =========================
% 双栏布局支持
% =========================
\usepackage{multicol}
\setlength{\columnsep}{0.8em}

% =========================
% 超链接
% =========================
\usepackage[hidelinks]{hyperref}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% =========================
% 排版优化
% =========================
\sloppy
\tolerance=1000
\emergencystretch=2em
\hyphenpenalty=10000
\exhyphenpenalty=10000

% =========================
% 自定义命令 - 技能标签
% =========================
\newcommand{\skillitem}[1]{%
  \textcolor{TextColor}{#1}%
}

% 公司职位行格式
\newcommand{\jobline}[3]{%
  {\small\bfseries\color{SecondaryColor}#1} \hfill {\footnotesize\color{LightGray}#2}\\[-2pt]
  {\footnotesize\color{TextColor}#3}%
}
