% =========================
% Resume Style Definition
% =========================

% ---------- Colors ----------
\usepackage{xcolor}
\definecolor{PrimaryColor}{HTML}{2E3A59} % 深蓝灰

% ---------- Page Layout ----------
\usepackage[
  top=1.5cm,
  bottom=1.5cm,
  left=2cm,
  right=2cm
]{geometry}

% ---------- Line Spacing ----------
\usepackage{setspace}
\setstretch{1.05}

% ---------- Section Style (H2 in Markdown) ----------
\usepackage{titlesec}

% Disable numbering
\setcounter{secnumdepth}{0}

% H1 (# Title) - Mapped to \section in article class
\titleformat{\section}
  {\centering\Huge\bfseries\color{PrimaryColor}} % Format
  {} % Label
  {0em} % Sep
  {} % Before-code

\titlespacing{\section}{0pt}{0pt}{15pt}

% H2 (## Section) - Mapped to \subsection in article class
\titleformat{\subsection}
  {\color{PrimaryColor}\scshape\large\bfseries} % Format
  {} % Label
  {0em} % Sep
  {} % Before-code
  [\color{PrimaryColor}\titlerule] % Horizontal rule below

\titlespacing{\subsection}{0pt}{10pt}{6pt}

% H3 (### Subsection) - Mapped to \subsubsection in article class
\titleformat{\subsubsection}
  {\color{PrimaryColor}\bfseries} % Format
  {} % Label
  {0em} % Sep
  {} % Before-code

\titlespacing{\subsubsection}{0pt}{6pt}{2pt}

% ---------- Item List ----------
\usepackage{enumitem}
\setlist[itemize]{
  leftmargin=1.2em,
  itemsep=2pt,
  topsep=2pt
}

% ---------- Links ----------
\usepackage[hidelinks]{hyperref}

% ---------- Paragraph ----------
\setlength{\parindent}{0pt}
