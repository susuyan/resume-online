% ===== Header =====
\newcommand{\Header}[4]{%
  % Name row: name left, contact right
  \begin{tabularx}{\textwidth}{@{}X r@{}}
    {\fontsize{21}{23}\selectfont\bfseries\sffamily\color{ink} #1} &
    \raisebox{5pt}{\fontsize{7.8}{10}\selectfont\sffamily\color{muted} #3}
  \end{tabularx}\par
  \vspace{0.05em}
  % Title
  {\fontsize{9}{11}\selectfont\sffamily\bfseries\color{hi} #2}\par
  \vspace{0.10em}
  % Accent rule
  {\color{hi}\rule{\textwidth}{1.8pt}}\par
  \vspace{0.18em}
  % Summary
  {\fontsize{8.5}{12}\selectfont\color{body} #4}\par
  \vspace{0.30em}
}

% ===== Entry Heading =====
\newcommand{\EntryHeading}[3]{%
  \begin{tabularx}{\textwidth}{@{}X r@{}}
    {\fontsize{9}{11}\selectfont\sffamily\bfseries\color{ink} #1%
      \ifx&#2&\else
        {\color{rule}\ \vrule width 0.5pt height 7pt depth 1pt\ }%
        {\normalfont\fontsize{8.5}{11}\selectfont\color{body} #2}%
      \fi} &
    {\fontsize{7.5}{9}\selectfont\sffamily\color{muted} #3}
  \end{tabularx}\par
  \vspace{0.01em}
}

% ===== Meta Line (role/tag below entry heading) =====

}
