% ===== Latin fonts — Source Sans 3 → TeX Gyre Heros → Helvetica Neue =====
\IfFontExistsTF{Source Sans 3}{
  \setmainfont{Source Sans 3}[
    UprightFont = *-Regular,
    BoldFont    = *-SemiBold,
    ItalicFont  = *-Italic
  ]
  \setsansfont{Source Sans 3}[
    UprightFont = *-Regular,
    BoldFont    = *-SemiBold
  ]
}{
  \IfFontExistsTF{TeX Gyre Heros}{
    \setmainfont{TeX Gyre Heros}
    \setsansfont{TeX Gyre Heros}
  }{
    \IfFontExistsTF{Helvetica Neue}{
      \setmainfont{Helvetica Neue}
      \setsansfont{Helvetica Neue}
    }{}
  }
}

% ===== CJK fonts — PingFang SC → Noto Sans CJK → WenQuanYi =====
\IfFontExistsTF{PingFang SC}{
  \setCJKmainfont{PingFang SC}[
    BoldFont   = PingFang SC Semibold,
    ItalicFont = PingFang SC Light
  ]
  \setCJKsansfont{PingFang SC}[
    BoldFont = PingFang SC Semibold
  ]
}{
  \IfFontExistsTF{Noto Sans CJK SC}{
    \setCJKmainfont{Noto Sans CJK SC}
    \setCJKsansfont{Noto Sans CJK SC}
  }{
    \IfFontExistsTF{WenQuanYi Micro Hei}{
      \setCJKmainfont{WenQuanYi Micro Hei}
      \setCJKsansfont{WenQuanYi Micro Hei}
    }{}
  }
}

% ===== Header =====
% Awesome-CV layout: name left (large), contact right (small, top-aligned)
% accent rule as visual separator between header and body
\newcommand{\Header}[4]{%
  \begin{tabularx}{\textwidth}{@{}X r@{}}
    {\fontsize{23}{25}\selectfont\bfseries\sffamily\color{darktext} #1} &
    \raisebox{7pt}{%
      \fontsize{7.8}{10}\selectfont\sffamily\color{graytext} #3}
  \end{tabularx}\par
  \vspace{0.04em}
  {\fontsize{9.5}{12}\selectfont\sffamily\bfseries\color{awesome} #2}\par
  \vspace{0.08em}
  {\color{awesome}\rule{\textwidth}{2pt}}\par
  \vspace{0.14em}
  {\fontsize{8.8}{13}\selectfont\color{bodytext} #4}\par
  \vspace{0.32em}
}

% ===== Entry heading — Awesome-CV cventry style =====
% negative pre-vspace pulls entry up; separator bar between title and org
\newcommand{\EntryHeading}[3]{%
  \vspace{-0.8mm}%
  \begin{tabularx}{\textwidth}{@{}X r@{}}
    {\fontsize{9.5}{11}\selectfont\sffamily\bfseries\color{darktext} #1%
      \ifx&#2&\else
        {\color{divcolor}\ \vrule width 0.5pt height 7pt depth 1.5pt\ }%
        {\fontsize{9}{11}\selectfont\normalfont\color{bodytext} #2}%
      \fi} &
    {\fontsize{7.8}{9}\selectfont\sffamily\color{lighttext} #3}
  \end{tabularx}\par
  \vspace{-0.5mm}
}
